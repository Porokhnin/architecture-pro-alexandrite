@startuml
actor       User       as user
participant Frontend    as front
participant Backend     as back
database    Cache       as cache
database    Database    as db

user -> front : Запрос об изменении статуса заказа
front -> back : Запрос об изменении статуса заказа

back -> db : Запрос об изменении статуса заказа
back <- db : Изменённый заказ

back -> cache : Запись измененного заказа в кэшем (AddOrUpdate)
back <- cache : Результат операции с кэшем

front <- back : Результат операции изменения статуса заказа
user <- front : Результат операции изменения статуса заказа
@enduml
